    TutorBookingOffer-->>LearnerBookingService: result

    LearnerBookingService->>LessonSnapshot: CreateFromLesson()

    LearnerBookingService->>Wallet: Get Escrow Wallet

    loop For each offered slot
        LearnerBookingService->>HeldFund: CreateForBooking()
        LearnerBookingService->>BookedSlot: Create()

    end

    LearnerBookingService->>TutorBookingOffer: Delete()

    LearnerBookingService->>IUnitOfWork: SaveAsync + Commit
    IUnitOfWork->>Database: Persist changes

    loop For each held fund
        LearnerBookingService->>HangfireConfig: ScheduleHeldFundReleaseJob()
    end

    LearnerBookingService->>NotificationService: SendToUsersAsync()

    LearnerBookingService-->>ILearnerBookingService: BookingResponse
    ILearnerBookingService-->>LearnerBookingController: BookingResponse
